+func (t *Tree) Reopen(id string) bool {
+	t.mu.Lock()
+	node, ok := t.nodes[id]
+	if !ok {
+		t.mu.Unlock()
+		return false
+	}
+	switch node.Status {
+	case StatusDone, StatusFailed, StatusCancelled, StatusUnverified:

+		return false

+	snapshot := *node
+	t.mu.Unlock()
+	t.persistNode(snapshot)
+	return true
